    % Корень (Игрок 1)
    \node (P1) at (0,0) [circle, fill, inner sep=1.5pt, label=above:{Игрок 1}] {};

    % Ветки Игрока 1
    \draw (P1) -- (-1.5,-1) node[midway, left, xshift=-2pt] {Молчать (C)};
    \draw (P1) -- (1.5,-1) node[midway, right, xshift=2pt] {Предать (D)};

    % Узлы Игрока 2
    \node (P2L) at (-1.5,-1) [circle, fill, inner sep=1.5pt] {};
    \node (P2R) at (1.5,-1) [circle, fill, inner sep=1.5pt] {};
    
    % Подпись Игрока 2 между узлами
    \node at (0,-0.8) {Игрок 2};

    % Информационное множество (пунктир)
    \draw[dashed, thick] (P2L) -- (P2R);

    % Ветки Игрока 2 (слева)
    \draw (P2L) -- (-2.25,-2.5) node[midway, left] {C} node[below] {(-1, -1)};
    \draw (P2L) -- (-0.75,-2.5) node[midway, right] {D} node[below] {(-3, 0)};

    % Ветки Игрока 2 (справа)

    \draw (P2R) -- (2.25,-2.5) node[midway, right] {D} node[below] {(-2, -2)};
